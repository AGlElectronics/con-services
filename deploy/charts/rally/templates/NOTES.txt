   initContainer applies database migrations:

     kubectl get pods -n {{ .Values.namespace }} -w

2. Check readiness (database reachable):

     kubectl exec -n {{ .Values.namespace }} deploy/rally-frontend -- {{ if .Values.internalTLS.enabled }}curl -s --cacert /etc/conservices-internal-ca/ca.crt https{{ else }}wget -qO- http{{ end }}://rally-backend:{{ .Values.backend.port }}/readyz

   Expect: {"status":"ok","database":"ok"}
