     kubectl get pods -n {{ .Values.namespace }} -w

     kubectl exec -n {{ .Values.namespace }} deploy/rally-frontend -- wget -qO- http://rally-backend:{{ .Values.backend.port }}/readyz

   Expect: {"status":"ok","database":"ok"}
