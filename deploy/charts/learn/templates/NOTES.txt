     kubectl get pods -n {{ .Values.namespace }} -w

     kubectl exec -n {{ .Values.namespace }} deploy/learn-frontend -- wget -qO- http://learn-backend:{{ .Values.backend.port }}/api/health

   Expect: {"status":"ok","database":"ok",...}
